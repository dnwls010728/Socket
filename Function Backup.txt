void Renderer::DrawBitmap(WindowsWindow* window, const Microsoft::WRL::ComPtr<ID2D1Bitmap>& kBitmap, const Math::Rect& kRect, const Math::Vector2& kPivot, float rotation_z, bool use_slice9, const Math::Rect& kSlice9Rect)
{
    D2DViewport* d2d_viewport = FindD2DViewport(window);
    if (!d2d_viewport) return;

    Microsoft::WRL::ComPtr<ID2D1Effect> effect;
    
    Microsoft::WRL::ComPtr<ID2D1DeviceContext> device_context;
    HRESULT hr = d2d_viewport->d2d_render_target.As(&device_context);
    if (FAILED(hr)) return;
    
    hr = device_context->CreateEffect(CLSID_D2D1ColorMatrix, &effect);
    if (FAILED(hr)) return;
    
    effect->SetInput(0, kBitmap.Get());
    effect->SetValue(D2D1_COLORMATRIX_PROP_COLOR_MATRIX, D2D1_MATRIX_5X4_F{
        1.f, 0.f, 0.f, 0.f,
        0.f, 0.f, 0.f, 0.f,
        0.f, 0.f, 1.f, 0.f,
        0.f, 0.f, 0.f, .1f,
        0.f, 0.f, 0.f, 0.f
    });
    
    D2D1_MATRIX_3X2_F transform;
    // d2d_viewport->d2d_render_target->GetTransform(&transform);
    device_context->GetTransform(&transform);
    
    D2D1_POINT_2F center = D2D1::Point2F(kPivot.x, kPivot.y);
    // d2d_viewport->d2d_render_target->SetTransform(D2D1::Matrix3x2F::Rotation(rotation_z, center));

    const float kWidth = kBitmap->GetSize().width;
    const float kHeight = kBitmap->GetSize().height;

    // D2D1_BITMAP_INTERPOLATION_MODE_NEAREST_NEIGHBOR - Point
    // D2D1_BITMAP_INTERPOLATION_MODE_LINEAR - Bilinear

    if (!use_slice9)
    {
        const float kWidthRatio = kRect.width / kWidth;
        const float kHeightRatio = kRect.height / kHeight;

        device_context->SetTransform(D2D1::Matrix3x2F::Scale(kWidthRatio, kHeightRatio) *
            D2D1::Matrix3x2F::Rotation(rotation_z, center));
        
        const D2D1_RECT_F kTempRect = D2D1::RectF(kRect.MinX() / kWidthRatio, kRect.MinY() / kHeightRatio, kRect.MaxX() / kWidthRatio, kRect.MaxY() / kHeightRatio);
        device_context->DrawImage(effect.Get(), {kTempRect.left, kTempRect.top});
    }
    else
    {
        device_context->SetTransform(D2D1::Matrix3x2F::Rotation(rotation_z, center));
        device_context->DrawImage(effect.Get(), {0.f, 0.f}, {0.f, 0.f, 10.f, 10.f});
        device_context->SetTransform(transform);

        device_context->SetTransform(D2D1::Matrix3x2F::Scale(180.f / kWidth, 1.f) *
            D2D1::Matrix3x2F::Rotation(rotation_z, center));
        device_context->DrawImage(effect.Get(), {10.f / (180.f / kWidth), 0.f}, {10.f, 0.f, 54.f, 10.f});
        device_context->SetTransform(transform);

        device_context->SetTransform(D2D1::Matrix3x2F::Rotation(rotation_z, center));
        device_context->DrawImage(effect.Get(), {134.f, 0.f}, {54.f, 0.f, 64.f, 10.f});
        device_context->SetTransform(transform);

        device_context->SetTransform(D2D1::Matrix3x2F::Scale(1.f, 50.f / kHeight) *
            D2D1::Matrix3x2F::Rotation(rotation_z, center));
        device_context->DrawImage(effect.Get(), {0.f, 10.f / (50.f / kHeight)}, {0.f, 10.f, 10.f, 54.f});
        device_context->SetTransform(transform);

        device_context->SetTransform(D2D1::Matrix3x2F::Scale(180.f / kWidth, 50.f / kHeight) *
            D2D1::Matrix3x2F::Rotation(rotation_z, center));
        device_context->DrawImage(effect.Get(), {10.f / (180.f / kWidth), 10.f / (50.f / kHeight)}, {10.f, 10.f, 54.f, 54.f});
        device_context->SetTransform(transform);

        device_context->SetTransform(D2D1::Matrix3x2F::Scale(1.f, 50.f / kHeight) *
            D2D1::Matrix3x2F::Rotation(rotation_z, center));
        device_context->DrawImage(effect.Get(), {134.f, 10.f / (50.f / kHeight)}, {54.f, 10.f, 64.f, 54.f});
        device_context->SetTransform(transform);

        device_context->SetTransform(D2D1::Matrix3x2F::Rotation(rotation_z, center));
        device_context->DrawImage(effect.Get(), {0.f, 43.f}, {0.f, 54.f, 10.f, 64.f});
        device_context->SetTransform(transform);

        device_context->SetTransform(D2D1::Matrix3x2F::Scale(180.f / kWidth, 1.f) *
            D2D1::Matrix3x2F::Rotation(rotation_z, center));
        device_context->DrawImage(effect.Get(), {10.f / (180.f / kWidth), 43.f}, {10.f, 54.f, 54.f, 64.f});
        device_context->SetTransform(transform);

        device_context->SetTransform(D2D1::Matrix3x2F::Rotation(rotation_z, center));
        device_context->DrawImage(effect.Get(), {134.f, 43.f}, {54.f, 54.f, 64.f, 64.f});
        device_context->SetTransform(transform);
    }
    
    device_context->SetTransform(transform);
}
